    Прошивка и протокол
    

    ESP32 отправляет по одной JSON-строке 4 раза в секунду (115200 бод) и отвечает на команду i2c_scan:
    

    // ESP32 → браузер (по строке на пакет)
{"t":42.4,"ta":-24.1,"i":780,"u":110.2,"acc":0.12}

// браузер → ESP32  (кнопка «Поиск устройств COM / I²C»)
{"cmd":"i2c_scan"}
// ESP32 → браузер: найденные адреса в десятичном виде
{"i2c":[90,64,104]}
    

    ПолеЧто означает
    

    tтемпература болтового стыка, °C (MLX90614)
    

    taтемпература воздуха, °C
    

    iток линии, А (INA219 × коэффициент ТТ)
    

    uнапряжение шины, кВ (необязательно: от ТН или шлюза МЭК 61850)
    

    accвибрация, g (MPU6050)
    

    

    // Arduino IDE: Adafruit MLX90614 / INA219 / MPU6050 + Wire
const float CT_RATIO = 240.0;            // ТТ 1200 А : 5 А
void loop() {
  if (Serial.available() && Serial.readStringUntil('\n').indexOf("i2c_scan") >= 0) scanI2C();
  float amps = ina.getCurrent_mA() / 1000.0 * CT_RATIO;
  Serial.printf("{\"t\":%.1f,\"ta\":%.1f,\"i\":%.0f,\"acc\":%.2f}\n",
                mlx.readObjectTempC(), mlx.readAmbientTempC(), amps, vibrationG());
  delay(250);
}

    Шлюз FastAPI (Python, WebSocket)
    

    Для работы без Web Serial и для нескольких рабочих мест: шлюз читает COM-порт и рассылает строки JSON по WebSocket.
    

    # pip install fastapi uvicorn pyserial
# uvicorn gateway:app --host 127.0.0.1 --port 8765
import asyncio, threading, serial
from contextlib import asynccontextmanager
from fastapi import FastAPI, WebSocket

PORT, BAUD = "/dev/ttyUSB0", 115200        # Windows: "COM3"
clients, loop = set(), None

def reader():
    with serial.Serial(PORT, BAUD, timeout=1) as ser:
        while True:
            line = ser.readline().decode("utf-8", "ignore").strip()
            if line.startswith("{"):
                asyncio.run_coroutine_threadsafe(fan_out(line), loop)

async def fan_out(line):
    for ws in list(clients):
        try: await ws.send_text(line)
        except Exception: clients.discard(ws)

@asynccontextmanager
async def lifespan(app):
    global loop
    loop = asyncio.get_running_loop()
    threading.Thread(target=reader, daemon=True).start()
    yield

app = FastAPI(lifespan=lifespan)

@app.websocket("/ws")
async def ws_endpoint(ws: WebSocket):
    # Production: check ws.headers.get("origin") and require a token
    await ws.accept(); clients.add(ws)
    try:
        while True: await ws.receive_text()
    except Exception: clients.discard(ws)
    

    Затем введите адрес ws://127.0.0.1:8765/ws в поле «Шлюз FastAPI» на вкладке «Подключение».
